sounddevice==0.5.1

# >=1.8.2, not 1.7.7: that version's Listener defines a `_handle` method
# that collides with Python 3.13's new Thread._handle attribute, crashing
# every keyboard event with "TypeError: '_thread._ThreadHandle' object is
# not callable". Fixed upstream by renaming to _handle_message. See
# PROGRESS.md's Phase 1 log — this one cost real debugging time.
pynput==1.8.2
pytest==8.3.3

# via `.venv/bin/python -c "from openwakeword.utils import download_models;
# download_models(model_names=['hey_jarvis_v0.1'])"` (downloads both tflite
